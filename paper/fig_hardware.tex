% fig_hardware.tex -- projected per-sample latency / energy bar chart.
% Two side-by-side log-scale bar charts.
%
% IMPORTANT: pgfplots `nodes near coords` on a log-scale y-axis displays
% the LOG of the value (not the raw value) -- e.g. log10(3.1) = 0.49 on a
% bar at y=3.1.  We therefore turn off `nodes near coords` and overlay
% explicit \node labels with the actual numeric values.
\begin{tikzpicture}

\begin{axis}[
  name=axLat,
  width=0.46\columnwidth, height=4.7cm,
  ybar, bar width=8pt,
  symbolic x coords={MN, CF, GPU},
  xtick=data,
  xticklabels={MNIST, CIFAR, GPU},
  xticklabel style={font=\scriptsize, rotate=0, anchor=north, yshift=-1pt},
  ymode=log, log basis y=10,
  ymin=1, ymax=200000,
  ytick={1, 10, 100, 1000, 10000, 100000},
  ylabel={Latency [ns]},
  tick label style={font=\scriptsize},
  ymajorgrids, grid style={gray!18, dashed},
  enlarge x limits=0.3,
  clip mode=individual,
  ylabel style={font=\scriptsize, yshift=-2pt},
  title style={yshift=-2pt, font=\scriptsize},
  title={Latency per sample},
]
\addplot[fill=pfPhotonic!70, draw=pfPhotonic!95]
  coordinates {(MN, 3.1) (CF, 94) (GPU, 12500)};
\node[font=\tiny, anchor=south, text=pfPhotonic!75!black]
  at (axis cs:MN, 3.1) {3.1};
\node[font=\tiny, anchor=south, text=pfPhotonic!75!black]
  at (axis cs:CF, 94) {94};
\node[font=\tiny, anchor=south, text=pfPhotonic!75!black]
  at (axis cs:GPU, 12500) {12{,}500};
\end{axis}

\begin{axis}[
  at={(axLat.outer east)}, anchor=outer west,
  xshift=4mm,
  width=0.46\columnwidth, height=4.7cm,
  ybar, bar width=8pt,
  symbolic x coords={MN, CF, GPU},
  xtick=data,
  xticklabels={MNIST, CIFAR, GPU},
  xticklabel style={font=\scriptsize, rotate=0, anchor=north, yshift=-1pt},
  ymode=log, log basis y=10,
  ymin=0.1, ymax=20000000,
  ytick={0.1, 1, 100, 10000, 1000000},
  ylabel={Energy [pJ]},
  tick label style={font=\scriptsize},
  ymajorgrids, grid style={gray!18, dashed},
  enlarge x limits=0.3,
  clip mode=individual,
  ylabel style={font=\scriptsize, yshift=-2pt},
  title style={yshift=-2pt, font=\scriptsize},
  title={Energy per sample},
]
\addplot[fill=pfNonlin!70, draw=pfNonlin!95]
  coordinates {(MN, 0.42) (CF, 24) (GPU, 3200000)};
\node[font=\tiny, anchor=south, text=pfNonlin!75!black]
  at (axis cs:MN, 0.42) {0.42};
\node[font=\tiny, anchor=south, text=pfNonlin!75!black]
  at (axis cs:CF, 24) {24};
\node[font=\tiny, anchor=south, text=pfNonlin!75!black]
  at (axis cs:GPU, 3200000) {3.2M};
\end{axis}

\end{tikzpicture}
