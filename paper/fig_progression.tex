  at (axis cs:Cut, 0.14) {\textbf{+0.128}: cost of\\full photon-native};

\node[font=\scriptsize\itshape, text=pfPhotonic!80!black, anchor=south, align=center]

\node[font=\scriptsize\itshape, text=pfNonlin!75!black, anchor=south]
  at (axis cs:Hybrid, 0.015) {beats baseline};
