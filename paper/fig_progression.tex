% fig_progression.tex -- architectural progression bar chart.
% Inline annotations removed to avoid overlap with bar value labels;
% caption explains the colour coding.  Target line drawn with \draw
% to avoid the auto-coord-label that `\addplot` triggers.
\begin{tikzpicture}
\begin{axis}[
  width=0.98\columnwidth,
  height=6.2cm,
  ybar,
  bar width=10pt,
  xlabel={Architectural milestone},
  ylabel={Gap vs.\ baseline},
  xlabel style={font=\footnotesize, yshift=2pt},
  ylabel style={font=\footnotesize},
  tick label style={font=\scriptsize},
  ymin=0, ymax=0.155,
  ytick={0, 0.025, 0.05, 0.075, 0.10, 0.125, 0.15},
  symbolic x coords={Cut, WaveOff, LearnAlpha, AddOnly, MZM, K2},
  xtick=data,
  xticklabels={Strict\\cut, Wave-\\length, Learn.\\$\alpha$, Add.\\$A_{\mathrm{ref}}$, MZM\\(K1), bs/lr\\(K2)},
  xticklabel style={rotate=0, anchor=north, align=center,
                    font=\tiny, yshift=-1pt},
  nodes near coords,
  nodes near coords style={font=\scriptsize, /pgf/number format/fixed,
                           /pgf/number format/precision=4},
  every node near coord/.append style={yshift=1.5pt},
  grid=major, grid style={gray!18, dashed},
  enlarge x limits=0.10,
  clip mode=individual,
  title={\scriptsize Photon-native progression: deletion $\to$ ceiling-break $\to$ target hit},
  title style={yshift=-2pt},
]

% +0.05 target line drawn with \draw to bypass `nodes near coords`
% (which would otherwise auto-label both endpoints with "0.05").
\draw[dashed, thick, red!55]
  (axis cs:Cut, 0.05) -- (axis cs:K2, 0.05);

% Strict cut -- red (cost of full photon-native deletion)
\addplot[fill=pfElec!80, draw=pfElec!95]
  coordinates {(Cut, 0.1241)};

% intermediate optimisations -- orange
\addplot[fill=pfPhotonic!75, draw=pfPhotonic!90]
  coordinates {(WaveOff, 0.1028) (LearnAlpha, 0.0953) (AddOnly, 0.0930)};

% K1 (MZM ceiling-break) -- darker orange
\addplot[fill=pfPhotonic!95, draw=pfPhotonic!95]
  coordinates {(MZM, 0.0710)};

% K2 (target hit) -- green
\addplot[fill=pfNonlin!85, draw=pfNonlin!95]
  coordinates {(K2, 0.0435)};

% +0.05 target annotation -- single label at far-left, anchored south-west
% above the dashed line endpoint, no overlap with any bar label.
\node[font=\scriptsize\itshape, anchor=south west, text=red!55!black,
      inner sep=1pt]
  at (axis cs:Cut, 0.052) {$+0.05$ target};

\end{axis}
\end{tikzpicture}
